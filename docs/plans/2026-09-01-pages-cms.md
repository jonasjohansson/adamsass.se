# Pages CMS Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Move the copy out of `index.html` into three YAML files Adam edits in Pages CMS, and generate `index.html` from them on every push.

**Architecture:** `content/*.yml` holds the copy. `build.mjs` renders `index.html` from a tagged template literal in `lib/template.mjs` and generates cover derivatives with `sharp`. A GitHub Action runs the build on pushes that touch content and commits the result back to `main`, where the existing legacy Pages deploy picks it up. Adam never sees markup.

**Tech Stack:** Node 22 (`node:test` built in, no test framework to install), `yaml`, `sharp`. No bundler, no framework, no static site generator.

**Design doc:** `docs/plans/2026-09-01-pages-cms-design.md`

`lib/markdown.mjs`:

```js
const ESCAPES = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' };

export function escapeHtml(text) {
  return String(text).replace(/[&<>"]/g, (c) => ESCAPES[c]);
}

function href(url) {
  if (!/^(https?:\/\/|mailto:|\/|assets\/)/.test(url)) {
    throw new Error(`unsupported URL: ${url}`);
  }
  return url;
}

// Inline only, on purpose. See CLAUDE.md: four type sizes, and nothing that
// explains the page. Block level markdown would let a heading in.
export function inline(text) {
  let out = escapeHtml(text);
  out = out.replace(/\[([^\]]+)\]\(([^)\s]+)\)/g, (_m, label, url) => {
    const target = href(url);
    const rel = target.startsWith('mailto:') ? '' : ' rel="noopener"';
    return `<a href="${target}"${rel}>${label}</a>`;
  });
  out = out.replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');
  out = out.replace(/\*([^*]+)\*/g, '<em>$1</em>');
  return out;
}
```

Note the order: links are resolved before emphasis so that `*` inside a link
label still becomes `<em>`, and the URL is taken before escaping can interfere.

const ID = /^[A-Za-z0-9_-]{11}$/;

export function youtubeId(input) {
  const value = String(input).trim();
  if (ID.test(value)) return value;

  const match = value.match(
    /(?:youtube\.com\/watch\?(?:.*&)?v=|youtu\.be\/|youtube(?:-nocookie)?\.com\/embed\/)([A-Za-z0-9_-]{11})/
  );
  if (!match) throw new Error(`not a YouTube URL: ${input}`);
  return match[1];
}
```

import { youtubeId } from './youtube.mjs';

const filled = (v) => typeof v === 'string' && v.trim() !== '';

export function validate({ site, records, videos, covers }) {
  const errors = [];

  if (!filled(site.email) || !site.email.includes('@')) {
    errors.push('site.email must be an email address');
  }
  if (!filled(site.city)) errors.push('site.city must not be empty');
  if (!/^https:\/\//.test(site.cv_url || '')) errors.push('site.cv_url must be an https URL');
  if (!filled(site.description)) errors.push('site.description must not be empty');
  if (!filled(site.short_description)) errors.push('site.short_description must not be empty');
  if (!Array.isArray(site.bio) || site.bio.length === 0 || !site.bio.every(filled)) {
    errors.push('site.bio must be a list of at least one paragraph');
  }
  if (!filled(site.quote?.text)) errors.push('site.quote.text must not be empty');
  if (!filled(site.quote?.source)) errors.push('site.quote.source must not be empty');

  if (!Array.isArray(records) || records.length === 0) {
    errors.push('there must be at least one record');
  } else {
    records.forEach((r, i) => {
      const at = `records[${i}]`;
      for (const field of ['title', 'band', 'label', 'cover']) {
        if (!filled(r[field])) errors.push(`${at}.${field} must not be empty`);
      }
      if (!Number.isInteger(r.year) || r.year < 1900 || r.year > 2100) {
        errors.push(`${at}.year must be a four digit year`);
      }
      if (!/^https:\/\//.test(r.link || '')) errors.push(`${at}.link must be an https URL`);
      if (filled(r.cover) && !covers.has(r.cover)) {
        errors.push(`${at}.cover is not on disk: ${r.cover}`);
      }
    });
  }

  if (!Array.isArray(videos) || videos.length === 0) {
    errors.push('there must be at least one video');
  } else {
    videos.forEach((v, i) => {
      if (!filled(v.title)) errors.push(`videos[${i}].title must not be empty`);
      try {
        youtubeId(v.url);
      } catch {
        errors.push(`videos[${i}].url is not a YouTube URL: ${v.url}`);
      }
    });
  }

  return errors;
}
```

**Step 4: Write `lib/images.mjs`**

```js
import { readdir, mkdir, stat } from 'node:fs/promises';
import path from 'node:path';
import sharp from 'sharp';

export const COVERS_SRC = 'assets/img/covers/src';
export const COVERS_OUT = 'assets/img/covers';
const WIDTHS = [300, 600];

export function coverSlug(file) {
  return path
    .basename(String(file), path.extname(String(file)))
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '');
}

export async function listCovers() {
  try {
    const files = await readdir(COVERS_SRC);
    return new Set(
      files
        .filter((f) => /\.(jpe?g|png)$/i.test(f))
        .map((f) => `${COVERS_SRC}/${f}`)
    );
  } catch {
    return new Set();
  }
}

async function isCurrent(out, srcMtime) {
  try {
    return (await stat(out)).mtimeMs >= srcMtime;
  } catch {
    return false;
  }
}

// One upload in, four files out. Adam should never have to think about this.
export async function buildCovers(records) {
  await mkdir(COVERS_OUT, { recursive: true });
  const written = [];

  for (const r of records) {
    const slug = coverSlug(r.cover);
    const srcMtime = (await stat(r.cover)).mtimeMs;

    for (const width of WIDTHS) {
      const jpg = `${COVERS_OUT}/${slug}-${width}.jpg`;
      const webp = `${COVERS_OUT}/${slug}-${width}.webp`;

      if (!(await isCurrent(jpg, srcMtime))) {
        await sharp(r.cover).resize(width, width, { fit: 'cover' })
          .jpeg({ quality: 82, mozjpeg: true }).toFile(jpg);
        written.push(jpg);
      }
      if (!(await isCurrent(webp, srcMtime))) {
        await sharp(r.cover).resize(width, width, { fit: 'cover' })
          .webp({ quality: 80 }).toFile(webp);
        written.push(webp);
      }
    }
  }

  return written;
}
```
